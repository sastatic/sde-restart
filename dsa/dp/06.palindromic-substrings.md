https://neetcode.io/problems/palindromic-substrings

```
class Solution:
    def countSubstrings(self, s: str) -> int:
        if len(s) < 2:
            return len(s)
        res = 0
        def expand(l, r):
            count = 0
            while (
                l >= 0
                and r < len(s)
                and s[l] == s[r]
            ):
                l -= 1
                r += 1
                count += 1
            return count
        for i in range(len(s)):
            res += expand(i, i)
            res += expand(i, i+1)
        return res
```
