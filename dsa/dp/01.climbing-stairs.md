```
class Solution:
    def climbStairs(self, n: int) -> int:
        dp = [0]*(n+1)
        def rec(n):
            if n<= 2:
                dp[n] = n
                return n
            if dp[n] == 0:
                dp[n] = rec(n-1) + rec(n-2)
            return dp[n]
        return rec(n)
```

class Solution:
    def climbStairs(self, n: int) -> int:
        if n <= 2:
            return n
        i = 1
        j = 2
        for _ in range(3, n+1):
            current = i+j
            i = j
            j = current
        return j
