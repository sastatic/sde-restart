Problem

https://neetcode.io/problems/house-robber-ii

```
class Solution:
    def rob(self, nums: List[int]) -> int:
        if len(nums) == 1:
            return nums[0]
        def rob_linear(houses):
            pre1, pre2 = 0, 0
            for house in houses:
                curr = max(pre1, pre2 + house)
                pre2 = pre1
                pre1 = curr
            return pre1
        return max(rob_linear(nums[:-1]), rob_linear(nums[1:]))
```
