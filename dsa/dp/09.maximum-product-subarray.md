Problem

https://neetcode.io/problems/maximum-product-subarray

max_ending_at[i] = max(nums[i], nums[i]*max_ending_at[i-1])
but this violets for below example.
ex: [-2, 3, -4]
so we track both max and min
max_prd[i] = max(nums[i], nums[i]*max_prd[i-1], nums[i]*min_prd[i-1])
min_prd[i] = min(nums[i], nums[i]*max_prd[i-1], nums[i]*min_prd[i-1])

```
class Solution:
    def maxProduct(self, nums: List[int]) -> int:
        n = len(nums)
        max_prd = [float("-inf")]*n
        min_prd = [float("inf")]*n
        max_prd[0] = nums[0]
        min_prd[0] = nums[0]
        for i in range(1, n):
            max_prd[i] = max(nums[i], nums[i]*max_prd[i-1], nums[i]*min_prd[i-1])
            min_prd[i] = min(nums[i], nums[i]*max_prd[i-1], nums[i]*min_prd[i-1])
        return max(max_prd)
```

```
class Solution:
    def maxProduct(self, nums: List[int]) -> int:
        cur_max = nums[0]
        cur_min = nums[0]
        result = nums[0]

        for x in nums[1:]:
            prev_max = cur_max
            cur_max = max(x, x * prev_max, x * cur_min)
            cur_min = min(x, x * prev_max, x * cur_min)

            result = max(result, cur_max)

        return result
```
