```
class Solution:
    def rob(self, nums: List[int]) -> int:
        n = len(nums)
        if n <= 1:
            return nums[0]
        dp = [0]*n
        dp[0] = nums[0]
        dp[1] = max(nums[0], nums[1])
        for i in range(2, n):
            dp[i] = max(dp[i-1], dp[i-2]+nums[i])
        return dp[n-1]
```

class Solution:
    def rob(self, nums: List[int]) -> int:
        pre1, pre2 = 0, 0
        for money in nums:
            curr = max(pre1, pre2 + money)
            pre2 = pre1
            pre1 = curr
        return pre1
