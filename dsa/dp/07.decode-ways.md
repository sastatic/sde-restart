class Solution:
    def numDecodings(self, s: str) -> int:
        if not s or s[0] == "0":
            return 0
        
        a, b = 1, 1
        for i in range(1, len(s)):
            curr = 0

            if s[i] != "0":
                curr += a
            
            c = int(s[i-1:i+1])
            if 10 <= c <= 26:
                curr += b
            
            b = a
            a = curr
        return a
