coins = [1, 3, 4]
target = 6
conditions:
if picked coin is 1 then result will be 1 + min_coin(5)
if picked coin is 3 then result will be 1 + min_coin(3)
if picked coin is 4 then result will be 1 + min_coin(2)

min_coin(6) = 1 + min(min_coin(5), min_coin(3), min_coin(2))

dp[x] = the minimum number of coins needed to make exactly amount x

dp[x] = 1 + min(dp[x - coin] for coin in coins)

```
class Solution:
    def coinChange(self, coins: List[int], amount: int) -> int:
        dp = [float("inf")] * (amount + 1)
        dp[0] = 0
        for current in range(1, amount+1):
            for coin in coins:
                if coin <= current:
                    dp[current] = min(dp[current], 1 + dp[current-coin])
        return -1 if dp[amount] == float("inf") else dp[amount]
```
