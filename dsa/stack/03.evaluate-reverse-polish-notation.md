## Problem Link

[Evaluate Reverse Polish Notation](https://neetcode.io/problems/evaluate-reverse-polish-notation)

```py
class Solution:
    def evalRPN(self, tokens: List[str]) -> int:
        stack = []
        operators = {"+", "-", "*", "/"}
        for c in tokens:
            if c not in operators:
                stack.append(int(c))
                continue
            b = stack.pop()
            a = stack.pop()
            if c == "+":
                stack.append(a+b)
            if c == "-":
                stack.append(a-b)
            if c == "*":
                stack.append(a*b)
            if c == "/":
                stack.append(int(a/b))
        return stack.pop()
```

```py
class Solution:
    def evalRPN(self, tokens: List[str]) -> int:
        def dfs():
            token = tokens.pop()
            if token not in "+-*/":
                return int(token)
            right = dfs()
            left = dfs()
            if token == '+':
                return left + right
            elif token == '-':
                return left - right
            elif token == '*':
                return left * right
            elif token == '/':
                return int(left / right)
        return dfs()
```
