class Solution:
    def reorderList(self, head: Optional[ListNode]) -> None:
        slow, fast = head, head.next
        while fast and fast.next:
            slow = slow.next
            fast = fast.next.next

        curr = slow.next
        slow.next = None
        prev = None
        while curr:
            next = curr.next
            curr.next = prev
            prev = curr
            curr = next

        second = prev
        first = head
        while second:
            l1, l2 = first.next, second.next
            first.next = second
            second.next = l1
            first, second = l1, l2
