class Solution:
    def copyRandomList(self, head: 'Optional[Node]') -> 'Optional[Node]':
        dummy = Node(0)
        curr = dummy
        temp = head
        oldNewNodeMap = {}
        while temp:
            new = Node(temp.val)
            oldNewNodeMap[temp] = new
            temp = temp.next
            curr.next = new
            curr = curr.next
        temp = head
        curr = dummy.next
        while temp:
            if temp.random:
                randomCorrespondence = oldNewNodeMap[temp.random]
                curr.random = randomCorrespondence
            curr = curr.next
            temp = temp.next

        return dummy.next
```

```
class Solution:
    def __init__(self):
        self.map = {}
    def copyRandomList(self, head: 'Optional[Node]') -> 'Optional[Node]':
        if head is None:
            return None
        if head in self.map:
            return self.map[head]
        copy = Node(head.val)
        self.map[head] = copy
        copy.next = self.copyRandomList(head.next)
        copy.random = self.map.get(head.random)
        return copy
