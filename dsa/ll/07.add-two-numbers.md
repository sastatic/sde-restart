class Solution:
    def addTwoNumbers(self, l1: Optional[ListNode], l2: Optional[ListNode]) -> Optional[ListNode]:
        dummy = ListNode()
        curr = dummy
        remainder = 0
        while l1 and l2:
            val = l1.val + l2.val + remainder
            remainder = val//10
            new = ListNode(val%10)
            curr.next = new
            curr = curr.next
            l1 = l1.next
            l2 = l2.next
        while l1:
            val = l1.val + remainder
            remainder = val//10
            new = ListNode(val%10)
            curr.next = new
            curr = curr.next
            l1 = l1.next
        while l2:
            val = l2.val + remainder
            remainder = val//10
            new = ListNode(val%10)
            curr.next = new
            curr = curr.next
            l2 = l2.next
        if remainder != 0:
            new = ListNode(remainder)
            curr.next = new
            curr = curr.next
        return dummy.next

class Solution:
    def add(self, l1, l2, carry):
        if not l1 and not l2 and carry == 0:
            return None
        v1 = l1.val if l1 else 0
        v2 = l2.val if l2 else 0
        carry, val = divmod(v1 + v2 + carry, 10)
        next_node = self.add(
            l1.next if l1 else None,
            l2.next if l2 else None,
            carry
        )
        return ListNode(val, next_node)
    def addTwoNumbers(self, l1: Optional[ListNode], l2: Optional[ListNode]) -> Optional[ListNode]:
        return self.add(l1, l2, 0)
