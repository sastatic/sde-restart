class Solution:
    def reverseBatch(self, start: ListNode, last: ListNode):
        prev = None
        while start != last:
            nxt = start.next
            start.next = prev
            prev = start
            start = nxt
        return prev

    def reverseKGroup(self, head: Optional[ListNode], k: int) -> Optional[ListNode]:
        if not head or k <= 1:
            return head
        dummy = ListNode(0, head)
        prevGroupLast = dummy
        while prevGroupLast.next:
            kth = prevGroupLast
            for _ in range(k):
                kth = kth.next
                if not kth:
                    return dummy.next
            groupStart = prevGroupLast.next
            nextGroupStart = kth.next
            reverseBatchStart = self.reverseBatch(groupStart, nextGroupStart)

            prevGroupLast.next = reverseBatchStart
            groupStart.next = nextGroupStart
            prevGroupLast = groupStart
        return dummy.next
