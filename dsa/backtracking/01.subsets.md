class Solution:
    def subsets(self, nums: List[int]) -> List[List[int]]:
        res = [[]]
        for num in nums:
            temp = []
            for a in res:
                temp.append(a+[num])
            res.extend(temp)
        return res

class Solution:
    def subsets(self, nums: List[int]) -> List[List[int]]:
        result = []
        current = []
        def backtrack(index):
            if index == len(nums):
                result.append(current.copy())
                return
            backtrack(index+1)
            current.append(nums[index])
            backtrack(index + 1)
            current.pop()
        backtrack(0)
        return result
