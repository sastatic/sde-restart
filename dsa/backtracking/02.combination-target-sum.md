```
class Solution:
    def combinationSum(self, nums: List[int], target: int) -> List[List[int]]:
        result = []
        current = []
        def backtrack(index, remain):
            if remain == 0:
                result.append(current.copy())
                return
            for i in range(index, len(nums)):
                if remain < nums[i]:
                    break
                current.append(nums[i])
                backtrack(i, remain-nums[i])
                current.pop()
        nums.sort()
        backtrack(0, target)
        return result
```
