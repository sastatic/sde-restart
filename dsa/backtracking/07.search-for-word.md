class Solution:
    def exist(self, board: List[List[str]], word: str) -> bool:
        rows, cols = len(board), len(board[0])
        def dfs(curr, r, c, path):
            if curr == word:
                return True
            if curr not in word:
                return False
            for rd, cd in ((0,1), (1,0), (0,-1), (-1,0)):
                rd, cd = r+rd, c+cd
                if rd < 0 or cd < 0 or rd >= rows or cd >= cols or (rd, cd) in path:
                    continue
                path.add((rd,cd))
                if dfs(curr + board[rd][cd], rd, cd, path):
                    return True
                path.remove((rd,cd))
            return False
        for r in range(rows):
            for c in range(cols):
                if dfs(board[r][c], r, c, {(r,c)}):
                    return True
        return False
