## Problem Link

[2094. Finding 3-Digit Even Numbers](https://leetcode.com/problems/finding-3-digit-even-numbers)

```
class Solution:
    def getEvenCombinations(self, a, b, c):
        combs = []
        if a%2 and b%2 and c%2:
            return []
        if a%2 == 0:
            if b:
                combs.append(b*100+c*10+a)
            if c:
                combs.append(c*100+b*10+a)
        if b%2 == 0:
            if a:
                combs.append(a*100+c*10+b)
            if c:
                combs.append(c*100+a*10+b)            
        if c%2 == 0:
            if a:
                combs.append(a*100+b*10+c)
            if b:
                combs.append(b*100+a*10+c)
        return combs

    def findEvenNumbers(self, digits: list[int]) -> list[int]:
        n = len(digits)
        res = set()
        for i in range(n):
            for j in range(i+1, n):
                for k in range(j+1, n):
                    combs = self.getEvenCombinations(digits[i], digits[j], digits[k])
                    res.update(combs)
        arr = list(res)
        arr.sort()
        return arr
```

Recursive Approach
```
class Solution:
    def findEvenNumbers(self, digits: list[int]) -> list[int]:
        res = set()
        n = len(digits)
        used = [False] * n

        def backtrack(depth: int, current_num: int):
            if depth == 3:
                res.add(current_num)
                return

            for i in range(n):
                if used[i]:
                    continue
                d = digits[i]
                if depth == 0 and d == 0:
                    continue
                if depth == 2 and d % 2 != 0:
                    continue
                used[i] = True
                backtrack(depth + 1, current_num * 10 + d)
                used[i] = False

        backtrack(0, 0)
        return sorted(list(res))
```

Optimized `O(N)`
```
from collections import Counter
class Solution:
    def findEvenNumbers(self, digits: list[int]) -> list[int]:
        res = []
        count = Counter(digits)
        def backtrack(depth, current_num):
            if depth == 3:
                res.append(current_num)
                return
            for d in range(10):
                if count[d] == 0:
                    continue
                if depth == 0 and d == 0:
                    continue
                if depth == 2 and d % 2 != 0:
                    continue
                count[d] -= 1
                backtrack(depth + 1, current_num * 10 + d)
                count[d] += 1
        backtrack(0, 0)
        return res
```
